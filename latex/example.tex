\documentclass{article}
\usepackage{aretino}

% Global defaults: CLI path and page-width rendering.
% On Linux, the CLI resolves text-font via fontconfig for text measurement;
% on other platforms, or for fully reproducible builds, set font-file too.
\aretinosetup{
  cli      = node ../node_modules/.bin/aretino,
  width-mm = 150,
  text-font=EB Garamond
}

\begin{document}
